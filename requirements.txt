# PocketTUI backend dependencies (app.py).
# Unpinned on purpose: the server uses only long-stable FastAPI/uvicorn API, and
# an install months from now should get a maintained release rather than a
# version that has since dropped support for the user's Python.
#
# starlette>=1.0 is not cosmetic: app.py passes allow_private_network=True to
# CORSMiddleware, which older starlette rejects with a TypeError at request time
# (every response becomes a 500). Naming it here turns that into an install-time
# resolution failure instead. starlette 1.x requires Python >=3.10, which is
# what really sets this project's minimum interpreter version.
fastapi>=0.100
starlette>=1.0
uvicorn[standard]>=0.23

# The primary speech recognizer (Parakeet-TDT). Pinned to a floor rather than
# left open because app.py builds the recognizer with modeling_unit="bpe", which
# older sherpa-onnx releases handle differently. Nothing here is required for
# the server to start: an install without the ONNX model, or without this wheel
# at all, falls back to the whisper.cpp build setup_voice.sh makes.
sherpa-onnx>=1.13.6

# Web Push for the "agent needs input" notifications (pulls in py-vapid and
# cryptography). Optional at runtime: app.py imports it lazily and answers
# push:false / 503 push_unavailable without it — the server must still boot
# and the ntfy transport still works on an install that skipped this wheel.
pywebpush>=2.0
